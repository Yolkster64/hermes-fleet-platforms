def conditional_corr(df, x, y, z, threshold=0.5):
    high = df[df[z] > df[z].median()]
    low  = df[df[z] <= df[z].median()]
    return high[x].corr(high[y]), low[x].corr(low[y])

        super().__init__()
        self.enc = nn.Sequential(nn.Linear(d, 64), nn.ReLU(), nn.Linear(64, 16))
